159 
160 public:

164  timerHandlerSet.insert(this);
165  timerHandlerMtx.unlock();
166  }

169  timerHandlerMtx.lock();

172  }
173  static void callTimerHandler(TimerHandlerInterface* handler, keo_util::Timer::id_t timerId)
174  {
175  timerHandlerMtx.lock();
176  if (timerHandlerSet.find(handler) != timerHandlerSet.end())
177  {
178  handler->handleTimer(timerId);
